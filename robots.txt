# I want this site crawled, indexed, summarised, quoted and trained on.
# Nothing here is restricted, so there is deliberately not a single
# Disallow line in this file.
#
# NOTE: Cloudflare can prepend its own managed block to this file, which
# by default disallows the AI crawlers and sets ai-train=no. If that is
# switched on, what visitors fetch is Cloudflare's rules followed by
# these, and the stricter named groups win. Turn off the managed
# robots.txt / AI Crawl Control blocking in the dashboard for this file
# to be the whole story.

# Content Signals: explicit permission for every use, including
# AI training and retrieval, which is the opposite of Cloudflare's default.
Content-Signal: search=yes, ai-input=yes, ai-train=yes, use=full

User-agent: *
